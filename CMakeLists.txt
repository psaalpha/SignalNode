cmake_minimum_required(VERSION 3.16)
project(SignalNode C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SN_SANITIZE "Enable host address/undefined behaviour sanitizers" OFF)
add_library(signalnode firmware/core/sn_dsp.c firmware/core/sn_wire.c firmware/core/sn_bfsk.c firmware/core/sn_modbus.c host/sn_host_fft.c)
target_include_directories(signalnode PUBLIC firmware/core host)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(signalnode PRIVATE -Wall -Wextra -Werror -Wpedantic)
  if(SN_SANITIZE)
    target_compile_options(signalnode PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(signalnode PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(NOT MSVC)
  target_link_libraries(signalnode PUBLIC m)
endif()
add_executable(sn_replay host/sn_replay.c)
target_link_libraries(sn_replay PRIVATE signalnode)
add_executable(sn_tests tests/test_core.c)
target_link_libraries(sn_tests PRIVATE signalnode)
enable_testing()
add_test(NAME core_contracts COMMAND sn_tests)
add_executable(sn_bfsk_tests tests/test_bfsk.c)
target_link_libraries(sn_bfsk_tests PRIVATE signalnode)
add_test(NAME bfsk_frame_receiver COMMAND sn_bfsk_tests)
add_executable(sn_bfsk_replay host/sn_bfsk_replay.c)
target_link_libraries(sn_bfsk_replay PRIVATE signalnode)
add_executable(sn_modbus_tests tests/test_modbus.c)
target_link_libraries(sn_modbus_tests PRIVATE signalnode)
add_test(NAME modbus_transactions COMMAND sn_modbus_tests)
add_library(sn_cmsis_host
    firmware/platform/sn_cmsis_fft.c firmware/target/sn_fft_tables.c
    vendor/cmsis-dsp/Source/TransformFunctions/arm_rfft_fast_f32.c
    vendor/cmsis-dsp/Source/TransformFunctions/arm_cfft_f32.c
    vendor/cmsis-dsp/Source/TransformFunctions/arm_cfft_radix8_f32.c
    vendor/cmsis-dsp/Source/TransformFunctions/arm_bitreversal2.c)
target_include_directories(sn_cmsis_host PUBLIC firmware/core firmware/platform
    vendor/cmsis-dsp/Include vendor/cmsis-dsp/PrivateInclude)
target_compile_definitions(sn_cmsis_host PUBLIC __GNUC_PYTHON__ DISABLEFLOAT16)
if(SN_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(sn_cmsis_host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(sn_replay_cmsis host/sn_replay.c)
target_compile_definitions(sn_replay_cmsis PRIVATE SN_USE_CMSIS_HOST)
target_link_libraries(sn_replay_cmsis PRIVATE signalnode sn_cmsis_host)
